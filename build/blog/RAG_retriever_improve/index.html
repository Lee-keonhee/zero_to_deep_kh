2. 하이브리드 검색 (Vector + BM25) ⭐⭐

from rank_bm25 import BM25Okapi

def hybrid_search(query, vectorstore, k=10):
    # 1. 벡터 검색
    vector_results = vectorstore.similarity_search(query, k=k*2)
    
    # 2. BM25 키워드 검색
    all_docs = vectorstore.get_all_documents()
    tokenized_docs = [doc.split() for doc in all_docs]
    bm25 = BM25Okapi(tokenized_docs)
    bm25_scores = bm25.get_scores(query.split())
    
    # 3. 점수 결합 (가중 평균)
    combined_results = []
    for doc in vector_results:
        vector_score = doc.metadata['score']
        bm25_score = bm25_scores[doc.id]
        final_score = 0.7 * vector_score + 0.3 * bm25_score
        combined_results.append((doc, final_score))
    
    # 4. 재정렬
    combined_results.sort(key=lambda x: x[1], reverse=True)
    return [doc for doc, score in combined_results[:k]]

효과:

  • Vector: 의미 유사도 (동의어 인식)
  • BM25: 정확한 키워드 매칭
  • 둘의 장점 결합!

4. 쿼리 확장 (Query Expansion) ⭐⭐

def expand_query(query):
    """쿼리에 동의어/관련어 추가"""
    
    prompt = f"""
    다음 쿼리를 검색에 최적화하기 위해 확장하세요:
    
    원본: {query}
    
    다음을 포함하여 확장하세요:
    1. 동의어 (예: "사업" → "프로젝트", "과제")
    2. 관련 용어 (예: "보안" → "정보보안", "데이터 보호")
    3. 약어 풀이 (예: "AI" → "인공지능")
    
    확장된 쿼리만 출력하세요.
    """
    
    expanded = llm.generate(prompt)
    return expanded

# 사용
original_query = "외부 용역 보안 특약"
expanded_query = expand_query(original_query)

results = vectorstore.search(expanded_query)

5. Hypothetical Document Embeddings (HyDE) ⭐⭐⭐

def hyde_search(query, vectorstore, k=10):
    """가상 답변을 만들어서 검색"""
    
    # 1. LLM에게 가상 답변 생성 요청
    prompt = f"""
    다음 질문에 대한 답변이 문서에 어떻게 쓰여있을지 예상해서 작성하세요:
    
    질문: {query}
    
    예상 답변 (2-3문장):
    """
    
    hypothetical_answer = llm.generate(prompt)
    
    # 2. 가상 답변으로 검색 (질문 대신!)
    results = vectorstore.similarity_search(hypothetical_answer, k=k)
    
    return results

왜 효과적?

질문: "외부 용역 보안 특약이 뭐야?"
→ 임베딩: [0.2, 0.5, ...] (질문 형태)

문서: "외부 용역 보안 특약은 계약 시 ..."
→ 임베딩: [0.3, 0.4, ...] (서술 형태)

→ 유사도 낮음 ❌

---

HyDE 가상 답변: "외부 용역 보안 특약은 계약 시 준수해야 할..."
→ 임베딩: [0.29, 0.41, ...] (서술 형태)

문서: "외부 용역 보안 특약은 계약 시 ..."
→ 임베딩: [0.3, 0.4, ...] (서술 형태)

→ 유사도 높음 ✅

6. Parent Document Retriever ⭐⭐

# 작은 청크로 검색하되, 큰 청크 반환
class ParentDocumentRetriever:
    def __init__(self, vectorstore):
        # 작은 청크 (검색용)
        self.small_chunks = create_chunks(docs, size=300)
        
        # 큰 청크 (LLM에 전달용)
        self.large_chunks = create_chunks(docs, size=1000)
        
        # 매핑 관계 저장
        self.chunk_mapping = {}  # small_id → large_id
    
    def search(self, query, k=5):
        # 1. 작은 청크로 검색 (정확도 높음)
        small_results = self.vectorstore.search(query, k=k)
        
        # 2. 해당하는 큰 청크 반환 (맥락 풍부)
        large_results = []
        for small in small_results:
            large_id = self.chunk_mapping[small.id]
            large_results.append(self.large_chunks[large_id])
        
        return large_results

# 개선: 유연한 필터
def flexible_filter(filters):
    """부분 매칭 허용"""
    where_clause = {}
    
    for key, values in filters.items():
        if key == '발주 기관':
            # OR 조건 + 부분 매칭
            where_clause['$or'] = [
                {key: {'$regex': f'.*{v}.*'}} 
                for v in values
            ]
    
    return where_clause

# 검색
results = vectorstore.search(
    query,
    where=flexible_filter({'발주 기관': ['고려대']})
)
# → "고려대학교", "고려대", "KU" 모두 매칭!
