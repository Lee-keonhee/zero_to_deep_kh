파이토치 QAT 경량화

양자화 연산 예시

양자화를 고려한 학습 방식

  • 일반적인 순전파, 오차역전파 학습 방식에 가중치와 활성화 값에 가짜 양자화(fake quantization) 연산을 삽입
  • 순전파 진행 시 원본 가중치에 양자화 → 역양자화를 적용하여 양자화 오차를 발생시킴
  • 양자화 오차가 고려된 손실로 부터 역전파를 진행하여 양자화 오차를 고려한 가중치 업데이트
  • 이를 통해 모델은 양자화 환경에 적합한 파라미터를 학습
import torch  
  
def symmetric_quantization(x, num_bits=8):  
    # 대칭적 양자화  
    qmin = -2 ** (num_bits - 1)  
    qmax = 2 ** (num_bits - 1) - 1  
  
    max_val = torch.max(torch.abs(x))  
    scale = max_val / qmax  
  
    q_x = torch.clamp(torch.round(x / scale), qmin, qmax)  
  
    return q_x.int(),  scale  
  
  
def asymmetric_quantization(x, num_bits=8):  
    # 비대칭적 양자화  
    qmin = -2 ** (num_bits - 1)  
    qmax = 2 ** (num_bits - 1) -1  
  
    min_val, max_val = x.min(), x.max()  
    print(max_val , min_val,qmax , qmin)  
    scale = (max_val - min_val) / (qmax - qmin)  
    zero_point = torch.round(qmin -min_val / scale).int()  
  
    q_x = torch.clamp(torch.round(x / scale) + zero_point, qmin, qmax)  
  
    return q_x.int(), scale, zero_point  
  
def dequantize_symmetric(quantized_tensor, scale):  
    # 대칭적 역 양자화  
    return quantized_tensor.float() * scale  
  
def dequantize_asymmetric(quantized_tensor, scale, zero_point):  
    # 비대칭적 역 양자화  
    return (quantized_tensor.float() - zero_point) * scale  
  
  
tensor_float32 = torch.tensor([1.25,2.67,3.08,4.12,3.02,2.01,0.89,5.45,3.76], dtype=torch.float32)  
print("원본 텐서:", tensor_float32, "타입:", tensor_float32.dtype)  
  
# Symmetric Quantization  
print("\n=== Symmetric Quantization ===")  
quantized_sym, scale_sym = symmetric_quantization(tensor_float32)  
dequantized_sym = dequantize_symmetric(quantized_sym, scale_sym)  
print("양자화 텐서:", quantized_sym)  
print("스케일:", scale_sym)  
print("역변환 텐서:", dequantized_sym)  
  
# Asymmetric Quantization  
print("\n=== Asymmetric Quantization ===")  
quantized_asym, scale_asym, zero_point = asymmetric_quantization(tensor_float32)  
dequantized_asym = dequantize_asymmetric(quantized_asym, scale_asym, zero_point)  
print("양자화 텐서:", quantized_asym)  
print("스케일:", scale_asym)  
print("제로 포인트:", zero_point)  
print("역변환 텐서:", dequantized_asym)

FX Graph 방식의 QAT 양자화

양자화 엔진 설정

FX Graph 처리된 모델의 양자화 학습을 진행하기 위해서는 각 모듈별로 양자화 공식이 지정된 QConfig를 세팅해야 합니다. 양자화 백엔드 엔진에는 이러한 QConfig가 기본적으로 구축이 되어있으며, 이를 그대로 활용하거나 직접 QConfig를 구축하여 QAT를 진행할 수 도 있습니다.

import platform  
engines = torch.backends.quantized.supported_engines  
print(f'지원되는 양자화 엔진:{engines}')  
print(f'현재 장치 아키텍처: {platform.machine()}')

지원되는 양자화 엔진:['qnnpack', 'onednn', 'x86', 'fbgemm']
현재 장치 아키텍처: x86_64

자동 QConfig 설정

기본적으로 설정한 엔진에 구축되어있는 qconfig를 전역에 설정하여 활용 합니다.

import torch  
import torchvision.models as models  
from torch import nn  
  
class WTCNNLogits(nn.Module):  
    def __init__(self, num_classes: int = 4):  
        super().__init__()  
        self.model = models.mobilenet_v3_small(  
            weights=models.MobileNet_V3_Small_Weights.IMAGENET1K_V1  
        )  
        in_features = self.model.classifier[0].in_features  
        self.model.classifier = nn.Identity()  
        self.dropout = nn.Dropout(0.3)  
        self.fc = nn.Linear(in_features, num_classes)  
  
    def forward(self, x):  
        x = self.model(x)  
        x = self.dropout(x)  
        x = self.fc(x)        # logits  
        return x

from torch.ao.quantization.quantize_fx import prepare_qat_fx, convert_fx  
from torch.ao.quantization import qconfig_mapping  
  
model_name = 'flower_qt'  
  
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")  
  
qconfig_mapping = qconfig_mapping.get_default_qconfig_mapping(backend='x86')  
  
model_fp32 = WTCNNLogits().eval().to(device)  
  
example_input = torch.randn(1, 3, 224, 224)  
qat_model = prepare_qat_fx(model_fp32, qconfig_mapping, example_inputs=(example_input.to(device),))  
print("[QAT] prepare_qat_fx done")

수동 QConfig 설정

직접 가중치와 활성화에 어떤 양자화 공식이 지정될지를 설정하여 QConfig를 구축합니다.

from torch.ao.quantization import get_default_qat_qconfig, QConfigMapping  
from torch.ao.quantization.observer import MovingAverageMinMaxObserver, PerChannelMinMaxObserver  
from torch.ao.quantization import QConfig, FakeQuantize  
  
model_name = 'flower_qt'  
  
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")  
  
# 직접 QConfig을 정의  
safe_qconfig = QConfig(  
    activation=MovingAverageMinMaxObserver.with_args(  
        dtype=torch.quint8, qscheme=torch.per_tensor_affine  
    ),  
    weight=PerChannelMinMaxObserver.with_args(  
        dtype=torch.qint8, qscheme=torch.per_channel_symmetric  
    )  
)  
  
# QConfigMapping 생성  
qconfig_mapping = QConfigMapping().set_global(safe_qconfig)  
  
model_fp32 = WTCNNLogits().eval().to(device)  
example_input = torch.randn(1, 3, 224, 224)  
# Pass example_inputs with the correct device  
qat_model = prepare_qat_fx(model_fp32, qconfig_mapping, example_inputs=(example_input.to(device),))  
print("[QAT] prepare_qat_fx done")

QAT 학습 및 양자화

QAT 학습 과정

기본적으로 QAT prepare가 된 모델은 옵저버를 활성화하여 학습 초기에 데이터의 통계(최대/최소)를 수집하여 Scale(스케일)과 Zero Point(제로 포인트)를 계산 합니다.

학습의 약 20%가 지난후 옵저버가 충분한 통계를 수집했다면 옵저버를 비활성 하고 통계수치를 고정하여 연산을 줄이고 안정적인 학습을 진행합니다.

from torch.ao.quantization import disable_observer  
from torch.ao.quantization.quantize_fx import prepare_qat_fx  
import os  
  
loss_fn = nn.CrossEntropyLoss()  
optimizer = torch.optim.Adam(qat_model.parameters(), lr=1e-3, weight_decay=1e-5)  
  
train_loss = []  
test_loss = []  
  
if not os.path.isdir('./checkpoint'):  
    os.mkdir('./checkpoint')  
  
# 학습 루프  
for epoch in range(6):  
    if epoch > 4:  
        # 두번의 에포크 이후 관측기(Observer)를 off        try:  
            qat_model.apply(disable_observer)  
            print("[QAT] Observers disabled (scales frozen).")  
        except Exception as e:  
            print(f"[QAT] disable_observer not applied: {e}")  
    t_acc = 0  
    qat_model.train()  
    for i, (images, labels) in enumerate(train_loader):  
        images = images.to(device)  
        labels = labels.to(device)  
        outputs = qat_model(images)  
        loss = loss_fn(outputs, labels)  
        optimizer.zero_grad()  
        loss.backward()  
        optimizer.step()  
  
        acc = (outputs.argmax(dim=1) == labels).sum() / len(labels)  
  
        t_acc += acc  
        if i % 10 == 0:  
            print('.',end='')  
  
    print(f"Epoch: {epoch}, Loss: {loss.item()}, Accuracy: {t_acc/(i+1)}")  
    train_loss.append(loss.item())  
  
    # 에포크 마다 체크포인트 저장(Fx 모델에서 불러와야함)  
    checkpoint_path = f"./checkpoint/ckpt_{model_name}{epoch}.pt"  
    torch.save(qat_model.state_dict(), checkpoint_path)  
  
    # 평가  
    v_acc = 0  
    qat_model.eval()  
    with torch.no_grad():  
        for i, (images, labels) in enumerate(val_loader):  
            images = images.to(device)  
            labels = labels.to(device)  
            outputs = qat_model(images)  
            loss = loss_fn(outputs, labels)  
            acc = (outputs.argmax(dim=1) == labels).sum() / len(labels)  
            v_acc += acc  
  
        print(f"Epoch: {epoch}, Val_Loss: {loss.item()}, Val_Accuracy: {v_acc/(i+1)}")  
        test_loss.append(loss.item())

PyTorch 2 Export (Pt2e)를 활용한 QAT

pt2e 방식 에선 prepare_qat_pt2e 함수에 백엔드 엔진을 설정하여 QAT 양자화를 적용할 수 있습니다.

pt2e 방식의 이점을 살려 QAT를 진행할 수 있으며 Executorch로 내보내어 런타임을 쉽게 구축할 수 있습니다.

QAT 모델 준비

일반 학습후 양자화 방식과 유사하게 export된 모델을 백엔드와 함께 prepare_qat_pt2e로 QAT가능한 prepare 모델을 생성합니다.

import torch  
  
model_name = 'flower_qt'  
device = 'cuda'  
# FP32 모델 준비  
model_fp32 = WTCNNLogits().to(device).train()  
# train 모드에서 더미 입력은 배치 2개 이상(동적 배치)  
example_input = torch.randn(2, 3, 224, 224, device=device)  
  
# torch.export로 ExportedProgram 생성  
exported = torch.export.export(model_fp32, (example_input,), strict=True).module()

from torchao.quantization.pt2e.quantize_pt2e import prepare_qat_pt2e  
from executorch.backends.xnnpack.quantizer.xnnpack_quantizer import (  
  get_symmetric_quantization_config,  
  XNNPACKQuantizer,  
)  
  
# is_qat 모드를 활성화 하여 qconfig 설정  
qconfig = get_symmetric_quantization_config(  
    is_per_channel=True,  
    is_qat=True  
)  
quantizer = XNNPACKQuantizer().set_global(qconfig)  
  
# QAT 모델 생성  
qat_model = prepare_qat_pt2e(exported, quantizer)  
print("[QAT] prepare_qat_pt2e done")

QAT 학습 과정

pt2e 모델은 .eval() 함수로 평가 모드로 변환하는 대신, 모델을 복사하고 move_exported_model_to_eval 함수를 통해 평가 모델을 구축해야 합니다.

import copy  
import os  
from torchao.quantization.pt2e import move_exported_model_to_eval, disable_observer  
  
loss_fn = nn.CrossEntropyLoss()  
optimizer = torch.optim.Adam(qat_model.parameters(),  
    lr=1e-3, weight_decay=1e-5)  
  
train_loss, test_loss = [], []  
os.makedirs('./checkpoint', exist_ok=True)  
  
  
for epoch in range(6):  
    if epoch > 4:  
        try:  
            qat_model.apply(disable_observer)  # PT2E 전용 observer freeze            print("[QAT] Observers disabled")  
        except Exception as e:  
            print(f"[QAT] freeze step skipped: {e}")  
  
    # --- Train (export 이후엔 .train() 호출 대신 학습 그래프를 그대로 사용) ---  
    t_acc = 0  
    for i, (images, labels) in enumerate(train_loader):  
        images = images.to(device)  
        labels = labels.to(device)  
        optimizer.zero_grad(set_to_none=True)  
        outputs = qat_model(images)  
        loss = loss_fn(outputs, labels)  
        loss.backward()  
        optimizer.step()  
  
        acc = (outputs.argmax(dim=1) == labels).sum() / len(labels)  
        t_acc += acc  
        if i % 10 == 0:  
            print('.', end='')  
    print(f"\nEpoch: {epoch}, Loss: {loss.item():.4f}, Accuracy: {(t_acc/(i+1)).item():.4f}")  
    train_loss.append(loss.item())  
  
    # 체크포인트 (재개 시 동일한 export→prepare 절차 필요)  
    torch.save(qat_model.state_dict(), f"./checkpoint/ckpt_{model_name}{epoch}.pt")  
  
    # --- Eval: 학습 그래프를 deepcopy → eval 그래프로 변환 후 평가 ---    v_acc = 0  
    eval_m = copy.deepcopy(qat_model)  
    move_exported_model_to_eval(eval_m)  # PT2E: model.eval() 대신 이 API 사용  
    with torch.no_grad():  
        for i, (images, labels) in enumerate(val_loader):  
            images = images.to(device)  
            labels = labels.to(device)  
            outputs = eval_m(images)  
            loss = loss_fn(outputs, labels)  
            acc = (outputs.argmax(dim=1) == labels).sum() / len(labels)  
            v_acc += acc  
    print(f"Epoch: {epoch}, Val_Loss: {loss.item():.4f}, Val_Accuracy: {(v_acc/(i+1)).item():.4f}")  
    test_loss.append(loss.item())

Executorch 내보내기

from executorch.backends.xnnpack.partition.xnnpack_partitioner import XnnpackPartitioner  
from executorch.exir import to_edge_transform_and_lower  
from torch.export import Dim, export  
  
int8_model.to('cpu')  
inputs = (torch.randn(2,3,224,224),)  
  
dynamic_shapes = {"x": {0: Dim("batch")}}  
  
exported_program = export(int8_model, inputs, dynamic_shapes= dynamic_shapes)  
executorch_program = to_edge_transform_and_lower(  
    exported_program,  
    partitioner = [XnnpackPartitioner()]  
).to_executorch()  
  
with open("model_int8.pte", "wb") as file:  
    file.write(executorch_program.buffer)
